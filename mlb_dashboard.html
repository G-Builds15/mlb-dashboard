// ───────────────────────────────────────────── // BUILD DROPDOWN FROM slate_data.js games object // Runs on page load — no hardcoded game keys // ───────────────────────────────────────────── function buildDropdown() { const sel = document.getElementById('gsel'); sel.innerHTML = ''; // Update header with today's date and game count from slate_data.js const meta = document.getElementById('hdr-meta'); if (meta && typeof games !== 'undefined') { const today = new Date().toLocaleDateString('en-US', {weekday:'short', month:'short', day:'numeric', year:'numeric'}); meta.textContent = today + ' · ' + Object.keys(games).length + ' Games'; } if (typeof games === 'undefined' || !games) { sel.innerHTML = ''; console.error('games not defined — slate_data.js failed to load'); return; } const keys = Object.keys(games); if (!keys.length) { sel.innerHTML = ''; return; } // Group by time period const afternoon = [], evening = [], late = []; keys.forEach(key => { const g = games[key]; const hour = parseInt((g.time || '').split(':')[0]); const isPM = (g.time || '').includes('PM'); const h24 = isPM && hour !== 12 ? hour + 12 : hour; if (h24 < 17) afternoon.push(key); else if (h24 < 21) evening.push(key); else late.push(key); }); const makeGroup = (label, arr) => { if (!arr.length) return; const grp = document.createElement('optgroup'); grp.label = label; arr.forEach(key => { const g = games[key]; const opt = document.createElement('option'); opt.value = key; opt.textContent = `${g.away} @ ${g.home} — ${g.time}`; grp.appendChild(opt); }); sel.appendChild(grp); }; makeGroup('── Afternoon ──', afternoon); makeGroup('── Evening ──', evening); makeGroup('── Late ──', late); // Load first game const firstKey = keys[0]; sel.value = firstKey; loadGame(firstKey); } // Auto-init on page load // Fire immediately if DOM already ready, otherwise wait if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', buildDropdown); } else { buildDropdown(); }
